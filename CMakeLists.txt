cmake_minimum_required(VERSION 3.25)
project(PulseMacro VERSION 1.4.6 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PULSE_CORE_ONLY "Build standalone replay tests without the Geode SDK" OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(pulse_core_tests tests/core_tests.cpp)
  target_include_directories(pulse_core_tests PRIVATE src)
  add_test(NAME pulse_core_tests COMMAND pulse_core_tests)
  add_executable(pulse_speedhack_tests tests/speedhack_tests.cpp)
  target_include_directories(pulse_speedhack_tests PRIVATE src)
  add_test(NAME pulse_speedhack_tests COMMAND pulse_speedhack_tests)
  add_executable(pulse_library_tests tests/library_tests.cpp)
  target_include_directories(pulse_library_tests PRIVATE src)
  add_test(NAME pulse_library_tests COMMAND pulse_library_tests "${CMAKE_CURRENT_BINARY_DIR}/library-test-files")
  add_executable(frame_window_tests FrameWindows/tests/core_tests.cpp)
  target_include_directories(frame_window_tests PRIVATE FrameWindows/src)
  add_test(NAME frame_window_tests COMMAND frame_window_tests)
  add_executable(frame_checkpoint_tests FrameWindows/tests/checkpoint_tests.cpp)
  target_include_directories(frame_checkpoint_tests PRIVATE FrameWindows/src)
  add_test(NAME frame_checkpoint_tests COMMAND frame_checkpoint_tests)
  add_executable(pulse_practice_tests tests/practice_tests.cpp)
  target_include_directories(pulse_practice_tests PRIVATE src FrameWindows/src)
  add_test(NAME pulse_practice_tests COMMAND pulse_practice_tests)
endif()
if(PULSE_CORE_ONLY)
  return()
endif()
if(NOT DEFINED ENV{GEODE_SDK})
  message(FATAL_ERROR "Set GEODE_SDK to a Geode 5.10.x SDK checkout.")
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
add_library(${PROJECT_NAME} SHARED src/main.cpp FrameWindows/src/main.cpp FrameWindows/src/Hud.cpp FrameWindows/src/Storage.cpp)
target_compile_definitions(${PROJECT_NAME} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(${PROJECT_NAME} PRIVATE src)
option(PULSE_INTEGRATION_TEST "Run a temporary local-level integration test on startup" OFF)
if(PULSE_INTEGRATION_TEST)
  target_compile_definitions(${PROJECT_NAME} PRIVATE PULSE_INTEGRATION_TEST=1)
endif()
setup_geode_mod(${PROJECT_NAME} DONT_INSTALL)
